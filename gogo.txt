using System;
using System.Data;
using System.Windows.Forms;

namespace LibraryManagement
{
    public partial class MainForm : Form
    {
        public MainForm()
        {
            InitializeComponent();
            LoadBooks();
        }

        private void LoadBooks()
        {
            DataTable booksTable = DatabaseHelper.GetAllBooks();
            lstBooks.DataSource = booksTable;
            lstBooks.DisplayMember = "Title";
        }

        private void btnAddBook_Click(object sender, EventArgs e)
        {
            AddEditBookForm addEditForm = new AddEditBookForm();
            if (addEditForm.ShowDialog() == DialogResult.OK)
            {
                DatabaseHelper.AddBook(
                    addEditForm.Book.Title,
                    addEditForm.Book.Authors,
                    addEditForm.Book.LibraryCode,
                    addEditForm.Book.YearPublished,
                    addEditForm.Book.Publisher,
                    addEditForm.Book.TotalCopies,
                    addEditForm.Book.AvailableCopies
                );
                LoadBooks();
            }
        }

        private void btnEditBook_Click(object sender, EventArgs e)
        {
            if (lstBooks.SelectedItem != null)
            {
                DataRowView selectedBookRow = (DataRowView)lstBooks.SelectedItem;

                Models.Book selectedBook = new Models.Book
                {
                    BookID = Convert.ToInt32(selectedBookRow["BookID"]),
                    Title = selectedBookRow["Title"].ToString(),
                    Authors = selectedBookRow["Authors"].ToString(),
                    LibraryCode = selectedBookRow["LibraryCode"].ToString(),
                    YearPublished = Convert.ToInt32(selectedBookRow["YearPublished"]),
                    Publisher = selectedBookRow["Publisher"].ToString(),
                    TotalCopies = Convert.ToInt32(selectedBookRow["TotalCopies"]),
                    AvailableCopies = Convert.ToInt32(selectedBookRow["AvailableCopies"])
                };

                AddEditBookForm editForm = new AddEditBookForm();
                editForm.Book = selectedBook;
                if (editForm.ShowDialog() == DialogResult.OK)
                {
                    DatabaseHelper.UpdateBook(
                        editForm.Book.BookID,
                        editForm.Book.Title,
                        editForm.Book.Authors,
                        editForm.Book.LibraryCode,
                        editForm.Book.YearPublished,
                        editForm.Book.Publisher,
                        editForm.Book.TotalCopies,
                        editForm.Book.AvailableCopies
                    );
                    LoadBooks();
                }
            }
            else
            {
                MessageBox.Show("Выберите книгу для редактирования.");
            }
        }

        private void btnSearch_Click(object sender, EventArgs e)
        {
            string searchTerm = txtSearch.Text.Trim();
            DataTable searchResult = DatabaseHelper.SearchBooks(searchTerm);
            lstBooks.DataSource = searchResult;
            lstBooks.DisplayMember = "Title";
        }

        private void btnLoanedBooks_Click(object sender, EventArgs e)
        {
            DataTable loansTable = DatabaseHelper.GetLoanedBooks();

            LoanedBooksForm loanedForm = new LoanedBooksForm();
            loanedForm.SetDataSource(loansTable);
            loanedForm.ShowDialog();
        }
    }
}
